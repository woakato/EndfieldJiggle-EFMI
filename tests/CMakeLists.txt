cmake_minimum_required(VERSION 3.20)
project(EndfieldJiggleEFMITests LANGUAGES CXX)
add_executable(loader_host loader_host.cpp)
target_compile_features(loader_host PRIVATE cxx_std_17)
target_compile_definitions(loader_host PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(loader_host PRIVATE user32)
set_property(TARGET loader_host PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
